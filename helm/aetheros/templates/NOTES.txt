  Getting the application URL
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

{{- if .Values.ingress.enabled }}
  AetherOS is exposed via Ingress at:

  {{- if .Values.ingress.tls.enabled }}
    https://{{ .Values.ingress.host }}/
    https://{{ .Values.ingress.host }}/api/health    ← Health check
    https://{{ .Values.ingress.host }}/dashboard     ← Frontend
  {{- else }}
    http://{{ .Values.ingress.host }}/
    http://{{ .Values.ingress.host }}/api/health     ← Health check
    http://{{ .Values.ingress.host }}/dashboard      ← Frontend

    NOTE: TLS is disabled. Enable it with:
      --set ingress.tls.enabled=true \
      --set ingress.tls.secretName=aetheros-tls
  {{- end }}

{{- else if contains "NodePort" .Values.service.type }}
  Get the NodePort URL by running:

    export NODE_PORT=$(kubectl get svc {{ include "aetheros.fullname" . }} \
      -n {{ .Release.Namespace }} \
      -o jsonpath='{.spec.ports[0].nodePort}')
    export NODE_IP=$(kubectl get nodes \
      -o jsonpath='{.items[0].status.addresses[0].address}')
    echo "http://$NODE_IP:$NODE_PORT"

{{- else if contains "LoadBalancer" .Values.service.type }}
  It may take a few minutes for the LoadBalancer IP to be assigned.
  Watch for it with:

    kubectl get svc {{ include "aetheros.fullname" . }} \
      -n {{ .Release.Namespace }} --watch

  Once EXTERNAL-IP is assigned:

    export SERVICE_IP=$(kubectl get svc {{ include "aetheros.fullname" . }} \
      -n {{ .Release.Namespace }} \
      -o jsonpath='{.status.loadBalancer.ingress[0].ip}')
    echo "http://$SERVICE_IP:{{ .Values.service.port }}"

{{- else }}
  The service type is ClusterIP. Port-forward to access locally:

    kubectl port-forward svc/{{ include "aetheros.fullname" . }} \
      8080:{{ .Values.service.port }} \
      -n {{ .Release.Namespace }}

  Then open: http://localhost:8080/

{{- end }}

  Health check
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Wait for all pods to be ready:

    kubectl rollout status deployment/{{ include "aetheros.fullname" . }} \
      -n {{ .Release.Namespace }}

  Watch pod status:

    kubectl get pods -n {{ .Release.Namespace }} -l \
      "app.kubernetes.io/name={{ include "aetheros.name" . }},\
app.kubernetes.io/instance={{ .Release.Name }}" --watch

  Tail logs:

    kubectl logs -n {{ .Release.Namespace }} \
      -l app.kubernetes.io/name={{ include "aetheros.name" . }} \
      -f --tail=100

  [ ] Replace all placeholder secrets in your values override file
      (AETHER_SECRET_KEY, AETHER_ONE_ZT_SECRET, DATABASE_URL, etc.)

  [ ] Run Alembic migrations from inside a pod:

        kubectl exec -it deploy/{{ include "aetheros.fullname" . }} \
          -n {{ .Release.Namespace }} -- \
          alembic upgrade head

  [ ] Set CORS_ORIGINS and FRONTEND_URL to your actual domain

  [ ] Configure your IdP for SSO / SAML if required

  [ ] For production: replace Helm-managed secrets with an external
      secret store (Vault, AWS Secrets Manager, Sealed Secrets)
